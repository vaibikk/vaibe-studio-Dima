}

def menu():
    rows = [[Btn(title, callback_data=key)] for key, (title, _) in SECTIONS.items()]
    rows.append([Btn("🌐 Наш сайт", url=SITE)])
    return Markup(rows)

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(HELLO, reply_markup=menu())

async def button(update: Update, context: ContextTypes.DEFAULT_TYPE):

    await q.answer()

        back = Markup([[Btn("⬅️ Назад", callback_data="back")]])
        await q.edit_message_text(f"{title}\n\n{text}", reply_markup=back)

app.add_handler(CallbackQueryHandler(button))

Бот-автоответчикОтвечает клиентам и пересылает их сообщения вам. 
import os
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes

TOKEN = os.environ["BOT_TOKEN"]
OWNER_ID = int(os.environ.get("OWNER_ID", "0"))  # твой Telegram ID (узнать у @userinfobot), чтобы получать сообщения клиентов

AUTO_TEXT = ("Спасибо за сообщение! 🙌\n"
             "Я сейчас не на связи, но отвечу в течение рабочего дня (пн–пт, 10:00–19:00).")

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("Здравствуйте! Напишите, что вам нужно, и я отвечу как только смогу.")

async def reply(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(AUTO_TEXT)
    if OWNER_ID:
        await context.bot.forward_message(chat_id=OWNER_ID,
                                          from_chat_id=update.effective_chat.id,
                                          message_id=update.message.message_id)

app = Application.builder().token(TOKEN).build()
app.add_handler(CommandHandler("start", start))
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, reply))
app.run_polling()
Бот-генератор паролейСоздаёт надёжный пароль на 8, 12 или 16 символов.

                               parse_mode="Markdown", reply_markup=KB)

Бот-калькулятор скидкиСчитает цену со скидкой и экономию. 
import os, re
from telegram import Update
from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes

TOKEN = os.environ["BOT_TOKEN"]

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        "🏷 Посчитаю цену со скидкой.\nНапиши цену и процент через пробел.\n\nНапример: 2500 20")

async def calc(update: Update, context: ContextTypes.DEFAULT_TYPE):
    nums = re.findall(r"\d+(?:[.,]\d+)?", update.message.text)
    if len(nums) != 2:
        await update.message.reply_text("Нужно два числа: цена и процент. Например: 2500 20")
        return
    price, percent = (float(n.replace(",", ".")) for n in nums)
    saved = price * percent / 100
    await update.message.reply_text(
        f"Цена со скидкой: {price - saved:.2f}\nЭкономия: {saved:.2f}")

app = Application.builder().token(TOKEN).build()
app.add_handler(CommandHandler("start", start))
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, calc))
app.run_polling()
